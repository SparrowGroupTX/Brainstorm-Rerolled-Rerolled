"""Publish a large committed update using small fast-forward pushes.

Keep existing commit history. Preload missing blobs through a temporary branch,
then publish the original branch and remove only the uploader's temporary refs.
No real upload occurs with the default 'plan' command.
"""
from __future__ import annotations

import argparse
from collections import defaultdict
from datetime import datetime, timedelta, timezone
import json
from pathlib import Path
import re
import subprocess
import sys
import time
from urllib.request import Request, urlopen
import uuid

ROOT = Path(__file__).resolve().parents[1]
MAX_BLOB = 100 * 1024 * 1024
BATCH_BYTES = 128 * 1024 * 1024


def git(root: Path, *args: str, data: bytes | None = None) -> bytes:
    return subprocess.check_output(["git", "-C", str(root), *args], input=data)


def batches(blobs: list[tuple[str, int]], limit: int) -> list[list[tuple[str, int]]]:
    if limit < 1:
        raise ValueError("Batch limit must be positive")
    result, batch, size = [], [], 0
    for sha, count in sorted(blobs):
        if count > limit:
            raise ValueError(f"Blob {sha} does not fit the batch limit")
        if batch and size + count > limit:
            result.append(batch)
            batch, size = [], 0
        batch.append((sha, count))
        size += count
    if batch:
        result.append(batch)
    return result


def advertised_cached_heads(root: Path, remote: str) -> list[str]:
    advertised = {row.split()[0] for row in
                  git(root, "ls-remote", "--heads", remote).splitlines()}
    cached = set(git(root, "for-each-ref", "--format=%(objectname)",
                     f"refs/remotes/{remote}/").splitlines())
    # Stale tracking refs must never count as data currently held by the server.
    return [sha.decode("ascii") for sha in sorted(advertised & cached)]


def missing_blobs(root: Path, target: str, remote: str,
                  known_heads: list[str] | None = None) -> list[tuple[str, int]]:
    # rev-list's negative traversal can still emit the same blob under a new
    # path. Payload trees use content-addressed names, so exclude by object ID
    # explicitly to make interrupted uploads resume without resending batches.
    if known_heads is None:
        held = git(root, "rev-list", "--objects", "--remotes=" + remote)
    elif known_heads:
        held = git(root, "rev-list", "--objects", "--stdin",
                   data=("\n".join(known_heads) + "\n").encode("ascii"))
    else:
        held = b""
    known = {row.split(b" ", 1)[0] for row in held.splitlines()}
    objects = git(root, "rev-list", "--objects", target)
    missing = [row.split(b" ", 1)[0] for row in objects.splitlines()
               if row.split(b" ", 1)[0] not in known]
    if not missing:
        return []
    rows = git(root, "cat-file", "--batch-check=%(objectname) %(objecttype) %(objectsize)",
               data=b"\n".join(missing) + b"\n")
    blobs = []
    for row in rows.splitlines():
        sha, kind, raw_size = row.decode("ascii").split()
        size = int(raw_size)
        if kind == "blob":
            if size > MAX_BLOB:
                raise ValueError(f"History still contains an ordinary blob above100MiB: {sha}")
            blobs.append((sha, size))
    return blobs


def payload_tree(root: Path, blobs: list[tuple[str, int]]) -> str:
    shards = defaultdict(list)
    for sha, _ in blobs:
        shards[sha[:2]].append(f"100644 blob {sha}\t{sha}\n")
    folders = []
    for prefix, rows in sorted(shards.items()):
        tree = git(root, "mktree", data="".join(sorted(rows)).encode("ascii")).decode().strip()
        folders.append(f"040000 tree {tree}\t{prefix}\n")
    payload = git(root, "mktree", data="".join(folders).encode("ascii")).decode().strip()
    return git(root, "mktree", data=f"040000 tree {payload}\tpayload\n".encode()).decode().strip()


def configuration(root: Path, remote: str | None, branch: str | None) -> tuple[str, str]:
    local = git(root, "symbolic-ref", "--short", "HEAD").decode().strip()
    remote = remote or git(root, "config", "--get", f"branch.{local}.remote").decode().strip()
    configured = git(root, "config", "--get", f"branch.{local}.merge").decode().strip()
    branch = branch or configured.removeprefix("refs/heads/")
    if not remote or remote == "." or remote.startswith("-"):
        raise ValueError("Select a named publication remote")
    git(root, "check-ref-format", "refs/heads/" + branch)
    return remote, branch


def publish_github_preload(root: Path, remote: str, ref: str, target: str, branch: str) -> bool:
    """Publish exact Git objects through GitHub after bounded blob preloads.

    Git can resend an already held blob when its containing tree differs. The
    GitHub tree API links held object IDs without uploading their bytes again.
    All returned tree/commit hashes must match local Git before updating the
    destination branch. Even an exact-tree bridge can resend held Git blobs.
    """
    url = git(root, "remote", "get-url", remote).decode().strip()
    match = re.fullmatch(r"https://github\.com/([\w.-]+/[\w.-]+?)(?:\.git)?", url)
    if not match:
        return False
    if not re.fullmatch(r"refs/heads/codex/upload-[0-9a-f]+-[0-9a-f]+", ref):
        raise ValueError("Tree assembly requires this uploader's temporary branch")
    git(root, "check-ref-format", "refs/heads/" + branch)
    held = advertised_cached_heads(root, remote)
    if missing_blobs(root, target, remote, held):
        raise ValueError("Complete all blob payloads before tree assembly")
    commits = git(root, "rev-list", "--reverse", target, "--not", *held).splitlines()
    if len(commits) > 32:
        raise ValueError("Tree assembly is limited to32 unpublished commits")
    tip = git(root, "ls-remote", "--heads", remote, ref).split()
    if len(tip) != 2 or tip[1].decode() != ref:
        raise ValueError("Temporary upload branch is not advertised")
    destination = git(root, "ls-remote", "--heads", remote, "refs/heads/" + branch).split()
    if len(destination) != 2:
        raise ValueError("Publication branch is not advertised")
    git(root, "merge-base", "--is-ancestor", destination[0].decode(), target)
    raw = subprocess.check_output(["git", "-C", str(root), "credential", "fill"],
                                  input=b"protocol=https\nhost=github.com\n\n")
    credentials = dict(row.split(b"=", 1) for row in raw.splitlines() if b"=" in row)
    secret = credentials.get(b"password")
    if not secret:
        raise ValueError("GitHub credential unavailable for tree assembly")
    base = "https://api.github.com/repos/" + match.group(1)
    request_count = [0]

    def api(path, body, method="POST"):
        if request_count[0] >= 80:
            raise ValueError("GitHub finalization exceeds80 metadata writes; branch unchanged")
        request_count[0] += 1
        encoded = json.dumps(body, separators=(",", ":")).encode("utf-8")
        if len(encoded) > 10 * 1048576:
            raise ValueError("Tree assembly request exceeds10MiB")
        request = Request(base + path, data=encoded, method=method, headers={
            "Accept": "application/vnd.github+json", "Content-Type": "application/json",
            "X-GitHub-Api-Version": "2026-03-10", "User-Agent": "Brainstorm-bounded-preload"})
        request.add_unredirected_header("Authorization", "Bearer " + secret.decode())
        with urlopen(request, timeout=60) as response:
            return json.load(response)

    trees = set()

    def identity(value):
        match = re.fullmatch(r"(.*) <([^<>]+)> (\d+) ([+-]\d{4})", value)
        if not match:
            raise ValueError("Unsupported Git commit identity")
        name, email, seconds, offset = match.groups()
        minutes = (int(offset[1:3]) * 60 + int(offset[3:])) * (-1 if offset[0] == "-" else 1)
        date = datetime.fromtimestamp(int(seconds), timezone(timedelta(minutes=minutes))).isoformat()
        return {"name": name, "email": email, "date": date}

    for commit in commits:
        commit = commit.decode()
        expected = git(root, "rev-parse", commit + "^{tree}").decode().strip()
        if expected not in trees:
            entries = []
            for row in git(root, "ls-tree", "-r", "-z", commit).split(b"\0"):
                if not row:
                    continue
                info, path = row.split(b"\t", 1)
                mode, kind, sha = info.decode("ascii").split()
                entries.append({"path": path.decode("utf-8"), "mode": mode, "type": kind, "sha": sha})
            partial = None
            for start in range(0, len(entries), 1000):
                body = {"tree": entries[start:start + 1000]}
                if partial:
                    body["base_tree"] = partial
                    time.sleep(1)
                result = api("/git/trees", body)
                partial = result.get("sha")
                if not isinstance(partial, str) or not re.fullmatch(r"[0-9a-f]{40}", partial):
                    raise ValueError("Invalid GitHub tree receipt")
                print(f"Tree metadata: {min(start + 1000, len(entries))}/{len(entries)} entries assembled", flush=True)
            if result.get("sha") != expected:
                raise ValueError("GitHub assembled tree differs from the frozen local tree")
            trees.add(expected)
        raw_commit = git(root, "cat-file", "commit", commit).decode("utf-8")
        raw_headers, message = raw_commit.split("\n\n", 1)
        fields = [line.split(" ", 1) for line in raw_headers.splitlines()]
        if any(key not in {"tree", "parent", "author", "committer"} for key, _ in fields):
            raise ValueError("Signed or extended commits need ordinary Git publication")
        parents = [value for key, value in fields if key == "parent"]
        values = dict(fields)
        result = api("/git/commits", {"tree": expected, "parents": parents, "message": message,
                     "author": identity(values["author"]), "committer": identity(values["committer"])})
        if result.get("sha") != commit or result.get("tree", {}).get("sha") != expected:
            raise ValueError("GitHub commit differs from the frozen local commit; branch unchanged")
    result = api("/git/refs/heads/" + branch, {"sha": target, "force": False}, "PATCH")
    if result.get("object", {}).get("sha") != target:
        raise ValueError("Publication reference verification failed")
    if git(root, "ls-remote", "--heads", remote, "refs/heads/" + branch).split()[0].decode() != target:
        raise ValueError("Published branch verification failed")
    git(root, "update-ref", "refs/remotes/" + remote + "/" + branch, target)
    print(f"Published exact commit through GitHub; {len(trees)} tree hashes verified without resending blobs", flush=True)
    return True


def publish(root: Path, remote: str, branch: str, limit: int = BATCH_BYTES) -> None:
    if git(root, "status", "--porcelain", "--untracked-files=normal").strip():
        raise ValueError("Commit the remaining changes first; this tool only uploads committed snapshots")
    target = git(root, "rev-parse", "HEAD").decode().strip()
    print(f"Publishing frozen commit {target} to {remote}/{branch}", flush=True)
    # Refresh advertised remote refs, without pruning or altering the user's branch.
    subprocess.run(["git", "-C", str(root), "fetch", "--no-auto-maintenance",
                    "--no-tags", remote], check=True)
    base = git(root, "rev-parse", f"refs/remotes/{remote}/{branch}").decode().strip()
    git(root, "merge-base", "--is-ancestor", base, target)
    grouped = batches(missing_blobs(root, target, remote,
                                   advertised_cached_heads(root, remote)), limit)
    name = "codex/upload-" + target[:12] + "-" + uuid.uuid4().hex[:10]
    ref = "refs/heads/" + name
    previous, uploaded = base, []
    created = False
    try:
        for number, batch in enumerate(grouped, 1):
            uploaded.extend(batch)
            tree = payload_tree(root, uploaded)
            message = f"Preload upload data {number}/{len(grouped)} for {target}\n"
            commit = git(root, "commit-tree", tree, "-p", previous,
                         data=message.encode("utf-8")).decode().strip()
            old = previous if created else "0" * len(commit)
            git(root, "update-ref", ref, commit, old)
            created = True
            print(f"Batch {number}/{len(grouped)}: at most {sum(n for _, n in batch) / 1048576:.1f} MiB of new blob data", flush=True)
            subprocess.run(["git", "-C", str(root), "-c", "core.compression=9",
                            "-c", "pack.compression=9", "push", "--progress", remote,
                            commit + ":" + ref], check=True)
            previous = commit
        api_published = created and publish_github_preload(root, remote, ref, target, branch)
        print("Publishing the original branch; commit history stays unchanged", flush=True)
        if not api_published:
            subprocess.run(["git", "-C", str(root), "-c", "core.compression=9",
                            "-c", "pack.compression=9", "push", "--progress", remote,
                            target + ":refs/heads/" + branch], check=True)
        advertised = git(root, "ls-remote", "--heads", remote, "refs/heads/" + branch)
        if advertised.split()[0].decode() != target:
            raise ValueError("Published branch verification failed; temporary upload refs retained")
    except BaseException:
        if created:
            print(f"Upload stopped; temporary ref retained for recovery: {remote}/{name}", file=sys.stderr)
        raise
    if created:
        cleanup = subprocess.run(["git", "-C", str(root), "push", remote, "--delete", name])
        if cleanup.returncode:
            print(f"Branch published, but temporary remote branch remains: {remote}/{name}", file=sys.stderr)
        else:
            git(root, "update-ref", "-d", ref, previous)
    print(f"Published {target}; no history rewrite or force push", flush=True)


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("command", choices=("plan", "push"), nargs="?", default="plan")
    parser.add_argument("--remote")
    parser.add_argument("--branch")
    parser.add_argument("--batch-mib", type=int, default=BATCH_BYTES // 1048576,
                        help="Maximum new uncompressed blob MiB per push (default:128)")
    args = parser.parse_args()
    try:
        if not 1 <= args.batch_mib <= 512:
            raise ValueError("Select a batch size between1 and512MiB")
        limit = args.batch_mib * 1048576
        remote, branch = configuration(ROOT, args.remote, args.branch)
        if args.command == "push":
            publish(ROOT, remote, branch, limit=limit)
        else:
            target = git(ROOT, "rev-parse", "HEAD").decode().strip()
            grouped = batches(missing_blobs(ROOT, target, remote), limit)
            print(f"Read-only plan for {remote}/{branch}: {len(grouped)} payload batches, then one final branch push.")
            print(f"Each batch has at most{args.batch_mib}MiB of new uncompressed blob data plus small Git metadata.")
            print("Plan uses current local remote refs; push refreshes them before uploading.")
            print("Commit all remaining changes, then run: python -B tools/push_in_batches.py push")
    except (OSError, ValueError, subprocess.CalledProcessError) as error:
        print(f"Batched upload stopped: {error}", file=sys.stderr)
        return 1
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
